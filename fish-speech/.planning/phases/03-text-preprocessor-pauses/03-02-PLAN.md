    ```python
    import time
    from fish_speech.utils.text_preprocessor import TextPreprocessor, PreprocessorConfig

    preprocessor = TextPreprocessor(PreprocessorConfig())

    # Test cases representing typical and adversarial inputs
    test_texts = [
        # Short text (typical chat message)
        "Hello, how are you today?",
        # Medium text with no punctuation (worst case for clause injection)
        "The quick brown fox jumped over the lazy dog and then ran through the forest and across the meadow and over the hills and past the river and into the village",
        # Long text with emotion tags (realistic dialogue)
        "[angry] You should never have done that to me! [sad] I trusted you completely and now everything is ruined. But maybe someday we can find a way to make things right again.",
        # Adversarial: 50+ words no punctuation
        " ".join(["word"] * 60),
        # Adversarial: 1KB of text
        "The cat sat on the mat. " * 40,
    ]

    for text in test_texts:
        times = []
        for _ in range(1000):
            start = time.perf_counter()
            preprocessor.preprocess(text)
            elapsed = (time.perf_counter() - start) * 1000  # ms
            times.append(elapsed)
        avg = sum(times) / len(times)
        p99 = sorted(times)[990]
        assert p99 < 10.0, f"P99 {p99:.3f}ms exceeds 10ms for text len={len(text)}"
        print(f"len={len(text):4d}  avg={avg:.3f}ms  p99={p99:.3f}ms")
    ```
